window.searchMap = async function() { const address = document.getElementById('mapSearch').value.trim(); if (!address) { alert('Vui lòng nhập địa chỉ cần tìm.'); return; } const status = document.getElementById('status'); status.innerText = '🔍 Đang tìm kiếm với OpenCage...'; // 👇 Dùng API key của bạn (lấy từ ảnh) const API_KEY = '0998742581b2444aafd5499e783a3588'; const url = `https://api.opencagedata.com/geocode/v1/json?q=${encodeURIComponent(address)}&key=${API_KEY}&limit=1&language=vi`; try { const response = await fetch(url); if (!response.ok) { throw new Error(`HTTP error! status: ${response.status}`); } const data = await response.json(); if (data.results && data.results.length > 0) { const result = data.results[0]; const lat = result.geometry.lat.toFixed(6); const lng = result.geometry.lng.toFixed(6); const latLng = `${lat}, ${lng}`; gpsText = latLng; document.getElementById('manualCoord').value = latLng; const displayName = result.formatted; // Địa chỉ đẹp const maxLen = getMaxLineLength(); const wrappedAddress = wrapAddress(displayName, maxLen); document.getElementById('address').value = wrappedAddress; status.innerText = '✅ Đã tìm thấy: ' + displayName; if (isLoaded) draw(); } else { status.innerText = '❌ Không tìm thấy địa điểm.'; alert('Không tìm thấy địa điểm phù hợp.'); } } catch (error) { console.error('Lỗi OpenCage:', error); status.innerText = '❌ Lỗi kết nối: ' + error.message; alert('Không thể kết nối đến OpenCage. Vui lòng thử lại sau.'); } };
